import 'dart:io';

String esc(String s) => s
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;');

String inline(String s) {
  var out = esc(s);
  out = out.replaceAllMapped(
      RegExp(r'`([^`]+)`'), (m) => '<code>${m[1]}</code>');
  out = out.replaceAllMapped(RegExp(r'\*\*([^*]+)\*\*'), (m) => '<b>${m[1]}</b>');
  return out;
}
